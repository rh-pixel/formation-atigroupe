function start(){
  const v=document.getElementById('nm').value.trim();
  if(!v) return;
  student=v; idx=0; answers=new Array(TOTAL).fill(null); renderQ();
}

function renderQ(){
  const q=QUESTIONS[idx], pct=Math.round(idx/TOTAL*100);
  const opts=q.o.map((o,i)=>`
    <button class="opt ${answers[idx]===i?'sel':''}" onclick="pick(${i})">
      <span class="mark">${String.fromCharCode(65+i)}</span><span>${esc(o)}</span>
    </button>`).join('');
  scr.innerHTML=`
    <div class="prog"><i style="width:${pct}%"></i></div>
    <div class="pinfo"><span>Question ${idx+1} / ${TOTAL}</span><span>${pct}%</span></div>

      <button class="btn ghost" onclick="prev()" ${idx===0?'disabled':''}>&#8592; Pr&eacute;c&eacute;dent</button>
      <button class="btn orange" onclick="next()" ${answers[idx]===null?'disabled':''}>
        ${idx===TOTAL-1?'Terminer et voir ma note':'Suivant &#8594;'}</button>
    </div>`;

function pick(i){answers[idx]=i;renderQ();}
function prev(){if(idx>0){idx--;renderQ();}}
function next(){if(answers[idx]===null)return;
  if(idx<TOTAL-1){idx++;renderQ();}else finish();}

function finish(){
  let score=0; QUESTIONS.forEach((q,i)=>{if(answers[i]===q.c)score++;});
  const pct=Math.round(score/TOTAL*100);
  const now=new Date();
  const date=now.toLocaleDateString('fr-FR')+' '+now.toLocaleTimeString('fr-FR').slice(0,5);
  const rec={nom:student,module:MOD_NUM,modtitre:MOD_TITRE,score:score,
             total:TOTAL,pct:pct,date:date,reponses:answers.slice()};
  const saved=saveOne(rec);
  renderResult(rec, saved);
}

function renderResult(rec, saved){
  const nom=rec.nom, score=rec.score, pct=rec.pct;
  let v,vs,col;
  if(pct>=80){v="Excellent !";vs="Le module est bien ma&icirc;tris&eacute;.";col="#1B7A2F";}

  else if(pct>=40){v="&Agrave; consolider";vs="Revoyez le module avant de continuer.";col="#F18F01";}
  else{v="&Agrave; revoir";vs="Une r&eacute;vision du module est recommand&eacute;e.";col="#B3261E";}
  const R=70,C=2*Math.PI*R,off=C*(1-score/TOTAL);
  const items=QUESTIONS.map((q,i)=>{
    const ok=answers[i]===q.c;
    const your=answers[i]==null?'<i>non r&eacute;pondu</i>':esc(q.o[answers[i]]);
    return `<div class="item ${ok?'ok':'ko'}">

function csvDL(filename, rows){
  const txt="\ufeff"+rows.map(r=>r.map(c=>'"'+(""+c).replace(/"/g,'""')+'"').join(';')).join("\r\n");
  const b=new Blob([txt],{type:'text/csv;charset=utf-8'});
  const a=document.createElement('a');
  a.href=URL.createObjectURL(b); a.download=filename;
  document.body.appendChild(a); a.click(); a.remove();
}
function slug(s){
  return (""+s).normalize('NFD').replace(/[\u0300-\u036f]/g,'')
    .replace(/[^a-zA-Z0-9]+/g,'_').replace(/^_+|_+$/g,'');
}
function dlFiche(rec){
  const rows=[];
  // Bloc identité (lignes clé/valeur, faciles à filtrer dans Excel)
  rows.push(["Nom et prénom", rec.nom]);
  rows.push(["Module", rec.module + " — " + rec.modtitre]);
  rows.push(["Note", rec.score + " / " + rec.total]);
  rows.push(["Pourcentage", rec.pct + "%"]);
  rows.push(["Date", rec.date]);
  rows.push([]);
  // Tableau détaillé : une ligne par question
  rows.push(["Nom","Module","N°","Question","Réponse du stagiaire",
             "Bonne réponse","Résultat"]);
  QUESTIONS.forEach((q,i)=>{
    const a=rec.reponses[i];
    const donnee = (a==null) ? "(non répondu)" : q.o[a];
    const ok = (a===q.c) ? "JUSTE" : "FAUX";
    rows.push([rec.nom, rec.module, (i+1), q.q, donnee, q.o[q.c], ok]);
  });
  const fname = slug(rec.module) + "__" + slug(rec.nom) + ".csv";
  csvDL(fname, rows);
}
